use aq_observation_admission::*;
use uuid::Uuid;
fn access(boot:u128)->Access{Access{node_id:Uuid::from_u128(1),boot_id:Uuid::from_u128(boot),accepted_through:0,max_pending_distance:256}}
fn request(id:u128,operation:Operation)->Request{Request{version:VERSION,operation_id:Uuid::from_u128(id),node_id:Uuid::from_u128(1),operation}}
fn main(){let p=request(10,Operation::Prepare{boot_id:Uuid::from_u128(2),store_id:Uuid::from_u128(3)});let prepared=apply(None,None,&p,&access(2),Some(Uuid::from_u128(4))).unwrap();let marker=prepared.reply.current_owner.marker.clone();let a=request(11,Operation::Activate{marker:marker.clone()});let active=apply(Some(&prepared.reply.current_owner),None,&a,&access(2),None).unwrap();let b=request(12,Operation::AdvanceBoot{marker:marker.clone(),expected_boot_id:Uuid::from_u128(2),new_boot_id:Uuid::from_u128(22),next_sequence:1});let advanced=apply(Some(&active.reply.current_owner),None,&b,&access(22),None).unwrap();let mut contradictory=advanced.reply.clone();contradictory.current_owner.current_boot_id=Uuid::from_u128(23);println!("same revision contradictory boot accepted: {}",contradictory.validate_against(&b).is_ok());let back=request(13,Operation::AdvanceBoot{marker,expected_boot_id:Uuid::from_u128(22),new_boot_id:Uuid::from_u128(2),next_sequence:1});println!("ABA to initial boot accepted: {}",apply(Some(&advanced.reply.current_owner),None,&back,&access(2),None).is_ok());}
